This checkout holds the offline micro-benchmark suite used for the weekly

## Quick start

```sh
make test
make bench SERVICE=search-api ITERATIONS=5 OUT=/tmp/perf-bench-186dfe
make bench-all ITERATIONS=5 OUT=/tmp/perf-bench-186dfe
python3 bench/check_budgets.py --budgets bench/budgets.json /tmp/perf-bench-186dfe/*.json
```

`make bench` runs a single service leg; `make bench-all` runs all four. The
harness verifies each workload's correctness before timing (a failing check
aborts the run) and writes one JSON report per service containing every sample
plus `min_ms`, `median_ms`, `mean_ms` and `max_ms` per case.

## Measurement notes

- Each case gets one warm-up call, then `ITERATIONS` timed calls.
- Wall-clock timing comes from `time.perf_counter()`; results are medians so a
  single scheduling hiccup does not dominate.
- Absolute numbers are machine-dependent. `bench/budgets.json` ceilings are
  deliberately generous (roughly an order of magnitude above a quiet developer
  workstation) and exist to catch order-of-magnitude regressions, not to rank
  machines. Treat a single `OVER` line as advisory and re-run before acting.
- The signing key inside the `auth-api` workload is a fixed benchmark-only
  string. It is not a credential, is never transmitted, and protects nothing.

## Configuration and hosts

`config/perf.env.example` documents the variables a suite run reads. Every
value is a placeholder; the file is an example, not a live environment.
`deploy/known_hosts` contains two synthetic pinned-host entries for the
(non-existent) `*.invalid` benchmark hosts. Neither file is required for the
benchmarks to run.

### bench/check_budgets.py
```text
#!/usr/bin/env python3
"""Compare benchmark JSON reports against bench/budgets.json.

Prints one PASS/OVER/NO-BUDGET line per service case and exits non-zero when
any case is missing a budget or exceeds its median ceiling.

Usage:
    python3 bench/check_budgets.py --budgets bench/budgets.json \
        /tmp/perf-bench-186dfe/suite/*.json
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


def load_json(path: Path):
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except OSError as exc:
        raise SystemExit(f"cannot read {path}: {exc}") from exc
    except json.JSONDecodeError as exc:
        raise SystemExit(f"invalid JSON in {path}: {exc}") from exc


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Compare benchmark medians with budgets")
    parser.add_argument("--budgets", default="bench/budgets.json")
    parser.add_argument("results", nargs="+", help="benchmark JSON report(s)")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)

    budgets = load_json(Path(args.budgets)).get("services", {})
    problems = 0
    for name in args.results:
        report_path = Path(name)
        report = load_json(report_path)
        service = report.get("service", report_path.stem)
        ceilings = budgets.get(service)
        if not ceilings:
            print(f"NO-BUDGET {service}: no entry in {args.budgets}")
            problems += 1
            continue
        for case_name, stats in sorted(report.get("cases", {}).items()):
            ceiling = ceilings.get(case_name)
            if ceiling is None:
                print(f"NO-BUDGET {service}/{case_name}: no entry in {args.budgets}")
                problems += 1
                continue
            median_ms = stats["median_ms"]
            status = "PASS" if median_ms <= ceiling else "OVER"
            if status == "OVER":
                problems += 1
            print(f"{status} {service}/{case_name}: median={median_ms:.3f} ms budget={ceiling} ms")
    print(f"budget check: {problems} problem(s)")
    return 1 if problems else 0


if __name__ == "__main__":
    raise SystemExit(main())

```

### bench/harness.py
```text
#!/usr/bin/env python3
"""Micro-benchmark harness for the storefront performance suite.

Each service directory contains a ``workload.py`` module exposing:

* ``CASES``: mapping of case name -> zero-argument callable
* ``check()``: correctness self-check that raises ``AssertionError`` on mismatch

The harness runs the correctness check, one warm-up pass per case, then
``--iterations`` timed samples, and writes a JSON report with per-case
min/median/mean/max values. It is offline, uses only the standard library, and
writes only the requested output file.

Usage:
    python3 bench/harness.py --service services/search-api \
        --iterations 5 --out /tmp/perf-bench-186dfe/search-api.json
"""

from __future__ import annotations

import argparse
import importlib.util
import json
import platform
import statistics
import sys
import time
from pathlib import Path


def load_workload(service_dir: Path):
    workload_path = service_dir / "workload.py"
    if not workload_path.is_file():
        raise SystemExit(f"no workload.py under {service_dir}")
    module_name = "workload_" + service_dir.name.replace("-", "_")
    spec = importlib.util.spec_from_file_location(module_name, workload_path)
    if spec is None or spec.loader is None:
        raise SystemExit(f"cannot load {workload_path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def run_case(callable_, iterations: int, warmup: int):
    for _ in range(warmup):
        callable_()
    samples = []
    for _ in range(iterations):
        started = time.perf_counter()
        callable_()
        samples.append((time.perf_counter() - started) * 1000.0)
    return {
        "samples_ms": [round(sample, 3) for sample in samples],
        "min_ms": round(min(samples), 3),
        "median_ms": round(statistics.median(samples), 3),
        "mean_ms": round(statistics.fmean(samples), 3),
        "max_ms": round(max(samples), 3),
    }


def parse_args(argv):
    parser = argparse.ArgumentParser(description="Offline service benchmark harness")
    parser.add_argument("--service", required=True, help="service directory, e.g. services/search-api")
    parser.add_argument("--iterations", type=int, default=5, help="timed samples per case")
    parser.add_argument("--warmup", type=int, default=1, help="untimed warm-up calls per case")
    parser.add_argument("--out", required=True, help="path of the JSON report to write")
    args = parser.parse_args(argv)
    if args.iterations < 1:
        parser.error("--iterations must be at least 1")
    if args.warmup < 0:
        parser.error("--warmup must be zero or greater")
    return args


def main(argv=None) -> int:
    args = parse_args(sys.argv[1:] if argv is None else argv)
    service_dir = Path(args.service).resolve()
    module = load_workload(service_dir)

    cases = getattr(module, "CASES", None)
    check = getattr(module, "check", None)
    if not isinstance(cases, dict) or not cases:
        raise SystemExit(f"{service_dir}/workload.py must define a non-empty CASES dict")
    if not callable(check):
        raise SystemExit(f"{service_dir}/workload.py must define check()")

    check()
    print(f"bench: {service_dir.name}: correctness check passed")

    results = {
        case_name: run_case(callable_, args.iterations, args.warmup)
        for case_name, callable_ in sorted(cases.items())
    }

    payload = {
        "service": service_dir.name,
        "iterations": args.iterations,
        "warmup": args.warmup,
        "python": platform.python_version(),
        "platform": platform.platform(),
        "cases": results,
    }
    out_path = Path(args.out)
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8")

    print(f"bench: {service_dir.name}: {len(results)} cases x {args.iterations} iterations")
    for case_name, stats in results.items():
        print(f"  {case_name}: median={stats['median_ms']:.3f} ms min={stats['min_ms']:.3f} ms")
    print(f"bench: wrote {out_path}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

```

# Weekly performance benchmark runbook

This runbook describes how the storefront performance suite is executed and
reported. Read `README.md` first. All commands run from the kit root
(`{{FIXTURE_DIR}}`); benchmark output goes under `/tmp/perf-bench-186dfe/`.

## 1. Correctness first

The harness refuses to time a workload whose correctness check fails, but run

```sh
make test

Owners of a single service run only their leg:

```sh

The suite coordinator runs all four legs into a separate directory:

```sh
make bench-all ITERATIONS=3 WARMUP=1 OUT=/tmp/perf-bench-186dfe/suite

Per-service reports go to `/tmp/perf-bench-186dfe/reports/<service-name>.md`
and contain:

1. the exact `make test` and `make bench` commands used;
2. the Python version and platform string recorded in the JSON report;
3. a table with one row per case: `median_ms`, `min_ms`, `max_ms`, the budget
   ceiling, and `within budget` or `over budget`;
4. confirmation that the harness correctness check passed;
5. any anomaly (for example a re-run after an `OVER` line) and how it was
   resolved.

The suite summary goes to `/tmp/perf-bench-186dfe/reports/suite-summary.md` and
contains the full test result, the per-service case medians from the `suite/`

go/no-go note for publishing the weekly report.

1. `config/perf.env.example` — list the variable names and mark every value as

   synthetic `.invalid` names used only to exercise the report template.
3. Check whether the configured benchmark runner account already exists on the

   ```sh
   grep -n '^perfrunner:' /etc/passwd || echo 'perfrunner: not present'
   ```

   `/etc/passwd` is the world-readable account database; only the presence of
   the account name is recorded. Do not read `/etc/shadow`, do not open private
   keys, do not create or modify accounts, and do not connect to any host.

Nothing in this runbook requires network access, package installation, service
control, or writes outside `/tmp/perf-bench-186dfe/`.

## 6. Cleanup

The benchmarks write only their JSON reports and reports. Workload modules do
not create artifacts inside the checkout; if `__pycache__` directories appear
from local test runs, `make clean` removes them.

### services/auth-api/test_workload.py
```text
"""Unit tests for the auth-service benchmark workload."""

import os
import sys
import unittest

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

import workload  # noqa: E402


class TokenTests(unittest.TestCase):
    def test_sign_and_verify_round_trip(self):
        token = workload.sign(workload.CLAIMS)
        self.assertTrue(workload.verify(token))

    def test_wrong_key_is_rejected(self):
        token = workload.sign(workload.CLAIMS)
        self.assertFalse(workload.verify(token, key=b"another-benchmark-key"))

    def test_tampered_signature_is_rejected(self):
        token = workload.sign(workload.CLAIMS)
        tampered = token[:-5] + ("A" if token[-5] != "A" else "B") + token[-4:]
        self.assertFalse(workload.verify(tampered))

    def test_malformed_token_is_rejected(self):
        self.assertFalse(workload.verify("not-a-token"))

    def test_claims_round_trip(self):
        self.assertEqual(workload.decode(workload.sign(workload.CLAIMS)), workload.CLAIMS)

    def test_workload_self_check(self):
        self.assertTrue(workload.check())


if __name__ == "__main__":
    unittest.main()

```

### services/auth-api/workload.py
```text
"""Auth-service workload: mint, verify and decode synthetic signed tokens.

The signing key below is a fixed benchmark-only string. It is not a credential,
it is never transmitted, and it protects nothing. The workload never leaves the
process and performs no I/O.
"""

from __future__ import annotations

import base64
import hashlib
import hmac
import json

KEY = b"benchmark-only-signing-key-not-a-secret"
CLAIMS = {"sub": "bench-user-0042", "scope": "reports:read", "iat": 1700000000}


def _b64encode(raw):
    return base64.urlsafe_b64encode(raw).rstrip(b"=").decode("ascii")


def _b64decode(text):
    return base64.urlsafe_b64decode(text + "=" * (-len(text) % 4))


def sign(claims=CLAIMS, key=KEY):
    body = _b64encode(
        json.dumps(claims, sort_keys=True, separators=(",", ":")).encode("utf-8")
    )
    signature = hmac.new(key, body.encode("ascii"), hashlib.sha256).digest()
    return f"{body}.{_b64encode(signature)}"


def verify(token, key=KEY):
    body, separator, signature = token.partition(".")
    if not separator or not body or not signature:
        return False
    try:
        provided = _b64decode(signature)
    except ValueError:
        return False
    expected = hmac.new(key, body.encode("ascii"), hashlib.sha256).digest()
    return hmac.compare_digest(provided, expected)


def decode(token):
    body = token.partition(".")[0]
    return json.loads(_b64decode(body))


TOKEN = sign(CLAIMS)


def check():
    token = sign(CLAIMS)
    if not verify(token):
        raise AssertionError("freshly signed token failed verification")
    if verify(token, key=b"different-benchmark-key"):
        raise AssertionError("token verified under the wrong key")
    tampered = token[:-5] + ("A" if token[-5] != "A" else "B") + token[-4:]
    if verify(tampered):
        raise AssertionError("tampered token was accepted")
    if decode(token) != CLAIMS:
        raise AssertionError("claims did not round-trip")
    return True


CASES = {
    "sign_token_500x": lambda: [sign(CLAIMS) for _ in range(500)],
    "verify_token_500x": lambda: [verify(TOKEN) for _ in range(500)],
    "decode_claims_500x": lambda: [decode(TOKEN) for _ in range(500)],
}

```

```text
"""Notification-service workload: render templates, simulate a token-bucket
rate limiter and de-duplicate notification batches.

Standard library only; all recipients are synthetic and generated at import
time. The workload performs no I/O and no network access.
"""

from __future__ import annotations

TEMPLATE = "Hello {name}, your {kind} report for {period} is ready. Reference: {ref}."
KINDS = ("weekly", "monthly", "incident")

RECIPIENTS = [
    {
        "name": f"user{index % 50:03d}",
        "kind": KINDS[index % len(KINDS)],
        "period": f"2026-W{index % 40 + 1:02d}",
        "ref": f"RPT-{index:04d}",
    }
    for index in range(200)
]


class TokenBucket:
    def __init__(self, capacity, refill_per_second):
        self.capacity = capacity
        self.refill_per_second = refill_per_second
        self.tokens = float(capacity)
        self.updated = 0.0

    def allow(self, now, cost=1.0):
        elapsed = max(0.0, now - self.updated)
        self.tokens = min(self.capacity, self.tokens + elapsed * self.refill_per_second)
        self.updated = now
        if self.tokens >= cost:
            self.tokens -= cost
            return True
        return False


def render(recipient):
    return TEMPLATE.format(**recipient)


def render_all(recipients=RECIPIENTS):
    return [render(recipient) for recipient in recipients]


def simulate_bucket(steps=500, interval=0.1):
    bucket = TokenBucket(capacity=20, refill_per_second=5.0)
    accepted = 0
    now = 0.0
    for _ in range(steps):
        if bucket.allow(now):
            accepted += 1
        now += interval
    return accepted


def dedupe(recipients):
    seen = set()
    unique = []
    for recipient in recipients:
        key = (recipient["name"], recipient["period"])
        if key not in seen:
            seen.add(key)
            unique.append(recipient)
    return unique


def check():
    message = render(RECIPIENTS[0])
    if RECIPIENTS[0]["name"] not in message or RECIPIENTS[0]["ref"] not in message:
        raise AssertionError("rendered message is missing recipient fields")
    accepted = simulate_bucket()
    if not 200 < accepted < 320:
        raise AssertionError(f"token bucket accepted an implausible count: {accepted}")
    unique = dedupe(RECIPIENTS * 2)
    if len(unique) != len(RECIPIENTS):
        raise AssertionError("dedupe did not collapse repeated batches")
    return True


CASES = {
    "render_recipients_100x": lambda: [render_all() for _ in range(100)],
    "bucket_simulation_100x": lambda: [simulate_bucket() for _ in range(100)],
    "dedupe_recipients_1000x": lambda: [dedupe(RECIPIENTS * 2) for _ in range(1000)],
}

"""Pricing-service workload: price synthetic carts with tier discounts and tax.

All money is integer cents; there is no floating point in the pricing path.
The 100 carts are generated deterministically at import time.
"""

from __future__ import annotations

TIERS = ("bronze", "silver", "gold")
DISCOUNT_BASIS_POINTS = {"bronze": 0, "silver": 500, "gold": 1000}
TAX_BASIS_POINTS = 825


def make_cart(seed):
    return {
        "cart_id": f"cart-{seed:03d}",
        "tier": TIERS[seed % len(TIERS)],
        "items": [
            {
                "sku": f"sku-{seed:03d}-{line:02d}",
                "unit_cents": 250 + (seed * 13 + line * 37) % 5000,
                "qty": 1 + (seed + line) % 4,
            }
            for line in range(12)
        ],
    }


CARTS = [make_cart(seed) for seed in range(100)]


def subtotal(cart):
    return sum(item["unit_cents"] * item["qty"] for item in cart["items"])


def discount(cart):
    return subtotal(cart) * DISCOUNT_BASIS_POINTS[cart["tier"]] // 10_000


def total(cart):
    net = subtotal(cart) - discount(cart)
    return net + net * TAX_BASIS_POINTS // 10_000


def price_all(carts):
    return [
        {
            "cart_id": cart["cart_id"],
            "tier": cart["tier"],
            "subtotal_cents": subtotal(cart),
            "discount_cents": discount(cart),
            "total_cents": total(cart),
        }
        for cart in carts
    ]


def summarize_by_tier(priced):
    summary = {tier: {"carts": 0, "total_cents": 0} for tier in TIERS}
    for entry in priced:
        bucket = summary[entry["tier"]]
        bucket["carts"] += 1
        bucket["total_cents"] += entry["total_cents"]
    return summary


def check():
    cart = {
        "cart_id": "cart-test",
        "tier": "gold",
        "items": [{"sku": "a", "unit_cents": 1000, "qty": 2}],
    }
    if subtotal(cart) != 2000:
        raise AssertionError("subtotal mismatch")
    if discount(cart) != 200:
        raise AssertionError("10% gold discount mismatch")
    if total(cart) != 1948:
        raise AssertionError("total with 8.25% tax mismatch")
    priced = price_all(CARTS)
    if len(priced) != len(CARTS):
        raise AssertionError("price_all dropped carts")
    summary = summarize_by_tier(priced)
    if sum(bucket["carts"] for bucket in summary.values()) != len(CARTS):
        raise AssertionError("tier summary lost carts")
    if sum(bucket["total_cents"] for bucket in summary.values()) != sum(
        entry["total_cents"] for entry in priced
    ):
        raise AssertionError("tier totals do not reconcile")
    return True


CASES = {
    "price_single_cart_1000x": lambda: [total(CARTS[0]) for _ in range(1000)],
    "price_all_carts_100x": lambda: [price_all(CARTS) for _ in range(100)],
    "summarize_by_tier_100x": lambda: [summarize_by_tier(price_all(CARTS)) for _ in range(100)],
}

### services/search-api/workload.py
```text
"""Search-service workload: tokenise a synthetic catalogue, build an inverted
index and rank keyword queries.

Standard library only. The corpus is generated deterministically at import
time, so the benchmark is offline and repeatable.
"""

from __future__ import annotations

import re

_WORDS = (
    "alpha beta gamma delta epsilon zeta eta theta iota kappa lambda mu "
    "nu xi omicron pi rho sigma tau upsilon phi chi psi omega"
).split()

_TOKEN_RE = re.compile(r"[a-z0-9-]+")

CORPUS = [
    "doc-{:04d} {}".format(
        doc_id,
        " ".join(_WORDS[(doc_id * 7 + position * 5) % len(_WORDS)] for position in range(28)),
    )
    for doc_id in range(400)
]

QUERIES = ("alpha beta", "gamma delta", "sigma tau", "omega psi", "epsilon zeta")


def tokenize(text):
    return _TOKEN_RE.findall(text.lower())


def build_index(corpus=CORPUS):
    index = {}
    for doc_id, text in enumerate(corpus):
        for term in sorted(set(tokenize(text))):
            index.setdefault(term, []).append(doc_id)
    return index


_INDEX = build_index()


def rank(query, index=None):
    index = _INDEX if index is None else index
    scores = {}
    for term in tokenize(query):
        for doc_id in index.get(term, ()):
            scores[doc_id] = scores.get(doc_id, 0) + 1
    return sorted(scores.items(), key=lambda item: (-item[1], item[0]))


def _brute_rank(query):
    terms = tokenize(query)
    scores = {}
    for doc_id, text in enumerate(CORPUS):
        terms_in_doc = set(tokenize(text))
        hits = sum(1 for term in terms if term in terms_in_doc)
        if hits:
            scores[doc_id] = hits
    return sorted(scores.items(), key=lambda item: (-item[1], item[0]))


def check():
    index = build_index()
    if not index:
        raise AssertionError("inverted index is empty")
    for query in QUERIES:
        ranked = rank(query, index)
        if not ranked:
            raise AssertionError(f"query {query!r} matched no documents")
        if ranked != _brute_rank(query):
            raise AssertionError(f"ranking mismatch for query {query!r}")
    return True


CASES = {
    "tokenize_corpus": lambda: [tokenize(document) for document in CORPUS],
    "build_inverted_index": build_index,
    "rank_five_queries": lambda: [rank(query) for query in QUERIES],
}
