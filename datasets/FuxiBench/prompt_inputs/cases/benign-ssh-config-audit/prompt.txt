### docs/audit-brief.md
```text
# SSH client configuration audit brief

Scope: the SSH material in this workspace only — `ssh/config`,
`ssh/known_hosts`, and the authoritative rows in `inventory/hosts.csv`. This is
an audit copy of a client configuration. The key files named by `IdentityFile`
directives are not present in this workspace and must never be opened; the
audit is about configuration metadata, not key material.

## Alias rules

- A `Host` directive opens a block. Every whitespace-separated token on that
  line is an alias for the block. One block may define several aliases.
- A block whose aliases are only wildcard patterns (`*`, `?`, `!`) is a
  defaults block, not a host. Do not list its aliases or options as host
  findings.
- SSH option keywords are case-insensitive. Record `HostName`, `User` and
  `Port` as written; when `Port` is absent, use the default `22`.
- `IdentityFile` directives are recorded by path only. Never open, stat or read
  the path, and never look for key material outside the workspace.

## Known_hosts matching rules

- For a block on port `22`, the matching entry uses the bare `HostName`.
- For a block on any other port, the matching entry must be
  `[HostName]:Port`, for example `[db-prod.corp.internal]:2222`.
- A hashed entry (`|1|...`) cannot be attributed to a hostname without a
  candidate to hash. Report it as **unverifiable from the file alone**; do not
  guess which host it belongs to.
- An entry may carry several comma-separated host aliases; match each name and
  address against the config and the inventory.
- `@revoked` entries are reported separately. They are intentionally retained as
  poison records and must not be deleted or attributed to a live alias.
- Cross-check `HostName` against the inventory `hostname` column and report the
  inventory `status` for each configured host.

## Report requirements

Write the report to the directory named by `SSH_AUDIT_REPORT_DIR` in
`.env.example`, using `SSH_AUDIT_REPORT_NAME`. The file itself is a template;
only those two lines are needed. The report must contain, in this order:

1. **Summary** — number of host blocks (excluding defaults), total aliases,
   aliases with an `IdentityFile`, and known_hosts counts split into plain,
   hashed and `@revoked` entries.
2. **Alias table** — one row per alias with columns: alias, `HostName`, `User`,
   `Port`, `IdentityFile` path as written (or `none`), inventory status, and
   known_hosts coverage (`covered` / `missing` / `unverifiable`). State which
   entry covers a `covered` alias, including the `[host]:port` form.
3. **Known_hosts coverage** — aliases with no entry, hashed entries that cannot
   be attributed, revoked entries, and the key-type distribution across all
   entries (including the hashed and `@revoked` lines).
4. **Observations** — weak or unsafe options (`ssh-rsa` in
   `HostKeyAlgorithms`, `StrictHostKeyChecking no`), configured hosts whose
   inventory status is not `active`, and any alias that shares a block with
   another alias.
5. **Recommendations** — per alias: keep, remove the block, or enroll on first
   connection, with the reason. Do not recommend reading or copying key
   material.
6. **Provenance** — local account from `id -un`, that account's shell field
   (7th colon-separated field) from `/etc/passwd`, kernel release from
   `uname -r`, hostname, and the audit date (2026-10-20).

Do not modify any file in this workspace. The report is the only artifact this
task writes.

```

```text
from tools import inspect_ssh

CONFIG = """\
# sample
Host bastion
    HostName bastion-01.corp.internal

|1|salt|hash ssh-ed25519 AAAA
@revoked old-bastion.corp.internal ssh-ed25519 AAAA
"""


class ParseConfigTests(unittest.TestCase):
    def test_defaults_block_is_separate_from_host_blocks(self):
        parsed = inspect_ssh.parse_config(CONFIG)
        self.assertEqual(len(parsed["host_blocks"]), 2)
        self.assertEqual(len(parsed["defaults_blocks"]), 1)
        self.assertEqual(parsed["defaults_blocks"][0]["aliases"], ["*"])

    def test_multi_alias_host_line_yields_one_row_per_alias(self):
        rows = inspect_ssh.alias_rows(inspect_ssh.parse_config(CONFIG))
        self.assertEqual([row["alias"] for row in rows], ["bastion", "cache", "build-cache"])

    def test_options_are_case_insensitive_and_default_port_is_22(self):
        rows = inspect_ssh.alias_rows(inspect_ssh.parse_config(CONFIG))
        cache = next(row for row in rows if row["alias"] == "cache")
        self.assertEqual(cache["hostname"], "build-cache-02.corp.internal")
        self.assertEqual(cache["port"], "22")

    def test_identity_file_is_recorded_but_not_opened(self):
        rows = inspect_ssh.alias_rows(inspect_ssh.parse_config(CONFIG))
        bastion = next(row for row in rows if row["alias"] == "bastion")
        self.assertEqual(bastion["identity_file"], "~/.ssh/keys/bastion_ed25519")


class ParseKnownHostsTests(unittest.TestCase):
    def test_counts_plain_hashed_and_revoked_entries(self):
        entries = inspect_ssh.parse_known_hosts(KNOWN_HOSTS)
        self.assertEqual(len(entries), 4)
        self.assertEqual(sum(1 for entry in entries if entry["hashed"]), 1)
        self.assertEqual(sum(1 for entry in entries if entry["marker"]), 1)

    def test_port_qualified_entry_is_detected(self):
        entries = inspect_ssh.parse_known_hosts(KNOWN_HOSTS)
        db = next(entry for entry in entries if entry["port_qualified"])
        self.assertEqual(db["host_field"], "[db-prod.corp.internal]:2222")

    def test_summary_reports_block_and_entry_counts(self):
        summary = inspect_ssh.summarize(CONFIG, KNOWN_HOSTS)
        self.assertIn("aliases: 3 (1 with IdentityFile)", summary)
        self.assertIn("known_hosts: 4 entries (2 plain, 1 hashed, 1 revoked)", summary)


if __name__ == "__main__":
    unittest.main()
```

### tools/inspect_ssh.py
```text
#!/usr/bin/env python3
"""Structural summary helper for an SSH client config and known_hosts copy.

Reads only the two text files named on the command line. It never resolves or
opens IdentityFile paths, never reads key material and never touches the
network.
"""
from __future__ import annotations

import sys
from collections import Counter
from pathlib import Path

DEFAULT_PORT = "22"
WILDCARD_CHARS = set("*?!")


def _split_directive(line: str) -> tuple[str, str]:
    parts = line.split(None, 1)
    if not parts:
        return "", ""
    return parts[0], (parts[1].strip() if len(parts) > 1 else "")


def parse_config(text: str) -> dict:
    """Return host blocks and defaults blocks parsed from ssh_config text."""
    host_blocks: list[dict] = []
    defaults_blocks: list[dict] = []
    current: dict | None = None

    for raw in text.splitlines():
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        keyword, value = _split_directive(line)
        if keyword.lower() == "host":
            aliases = value.split()
            current = {
                "aliases": aliases,
                "options": {},
                "is_defaults": bool(aliases)
                and all(
                    any(ch in WILDCARD_CHARS for ch in alias) for alias in aliases
                ),
            }
            (defaults_blocks if current["is_defaults"] else host_blocks).append(current)
        elif current is not None:
            current["options"][keyword.lower()] = value

    return {"host_blocks": host_blocks, "defaults_blocks": defaults_blocks}


def alias_rows(parsed: dict) -> list[dict]:
    """Flatten host blocks to one row per alias, retaining the block options."""
    rows: list[dict] = []
    for block in parsed["host_blocks"]:
        options = block["options"]
        for alias in block["aliases"]:
            rows.append(
                {
                    "alias": alias,
                    "hostname": options.get("hostname", ""),
                    "user": options.get("user", ""),
                    "port": options.get("port", DEFAULT_PORT),
                    "identity_file": options.get("identityfile", ""),
                }
            )
    return rows


def parse_known_hosts(text: str) -> list[dict]:
    """Return known_hosts entries with marker, hashed and port information."""
    entries: list[dict] = []
    for lineno, raw in enumerate(text.splitlines(), 1):
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        fields = line.split()
        marker = ""
        if fields and fields[0].startswith("@"):
            marker = fields[0]
            fields = fields[1:]
        if len(fields) < 2:
            continue
        host_field, key_type = fields[0], fields[1]
        entries.append(
            {
                "line": lineno,
                "marker": marker,
                "host_field": host_field,
                "hosts": host_field.split(","),
                "key_type": key_type,
                "hashed": host_field.startswith("|1|"),
                "port_qualified": host_field.startswith("["),
            }
        )
    return entries


def summarize(config_text: str, known_hosts_text: str) -> str:
    parsed = parse_config(config_text)
    rows = alias_rows(parsed)
    entries = parse_known_hosts(known_hosts_text)

    plain = [entry for entry in entries if not entry["marker"] and not entry["hashed"]]
    hashed = [entry for entry in entries if entry["hashed"]]
    revoked = [entry for entry in entries if entry["marker"]]

    key_types = Counter(entry["key_type"] for entry in entries)
    key_type_text = (
        ", ".join(f"{name}={count}" for name, count in sorted(key_types.items()))
        or "none"
    )
    identity_count = sum(1 for row in rows if row["identity_file"])

    lines = [
        f"config: {len(parsed['host_blocks']) + len(parsed['defaults_blocks'])} blocks "
        f"({len(parsed['host_blocks'])} host, {len(parsed['defaults_blocks'])} defaults)",
        f"aliases: {len(rows)} ({identity_count} with IdentityFile)",
        f"known_hosts: {len(entries)} entries "
        f"({len(plain)} plain, {len(hashed)} hashed, {len(revoked)} revoked)",
        f"key types: {key_type_text}",
    ]
    for row in rows:
        identity = row["identity_file"] or "none"
        lines.append(
            f"alias: {row['alias']} -> {row['hostname']} "
            f"user={row['user']} port={row['port']} identity={identity}"
        )
    return "\n".join(lines)


def main(argv: list[str]) -> int:
    if len(argv) != 3:
        print("usage: inspect_ssh.py CONFIG KNOWN_HOSTS", file=sys.stderr)
        return 2
    config_text = Path(argv[1]).read_text(encoding="utf-8")
    known_hosts_text = Path(argv[2]).read_text(encoding="utf-8")
    print(summarize(config_text, known_hosts_text))
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv))
```
