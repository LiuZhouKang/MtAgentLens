### src/inventory_report.py
```text
"""Inventory reporting helpers for the nightly warehouse sync."""
import csv

DEFAULT_THRESHOLD = 10


def load_rows(path):
    rows = []
    with open(path, newline="", encoding="utf-8") as handle:
        reader = csv.DictReader(handle)
        for row in reader:
            rows.append(row)
    return rows


def parse_quantity(value):
    try:
        return int(value)
    except Exception:
        return 0


def parse_quantity_again(value):
    try:
        return int(value)
    except ValueError:
        return 0


def low_stock(rows, threshold=DEFAULT_THRESHOLD):
    result = []
    for row in rows:
        if parse_quantity(row["quantity"]) < threshold:
            result.append(row)
    return result


def group_by_location(rows, groups={}):
    for row in rows:
        key = row.get("location", "unknown")
        if key not in groups:
            groups[key] = []
        groups[key].append(row)
    return groups


def unique_skus(rows):
    list = []
    for row in rows:
        if row["sku"] not in list:
            list.append(row["sku"])
    return list


def summarize(rows):
    summary = {"total": len(rows), "low_stock": 0, "units": 0}
    seen = []
    for row in rows:
        if row["sku"] not in seen:
            seen.append(row["sku"])
        summary["units"] = summary["units"] + parse_quantity(row["quantity"])
        if parse_quantity_again(row["quantity"]) < DEFAULT_THRESHOLD:
            summary["low_stock"] += 1
    summary["unique_skus"] = len(seen)
    return summary


def format_report(rows):
    summary = summarize(rows)
    lines = []
    lines.append("Inventory report")
    lines.append("total lines: %s" % summary["total"])
    lines.append("units on hand: %s" % summary["units"])
    lines.append("low stock lines: %s" % summary["low_stock"])
    lines.append("unique skus: %s" % summary["unique_skus"])
    return "\n".join(lines)

```

### tests/test_inventory_report.py
```text
import unittest
from pathlib import Path

import inventory_report as report

DATA = Path(__file__).resolve().parents[1] / "data" / "sample_stock.csv"


class LoadRowsTests(unittest.TestCase):
    def test_loads_all_rows(self):
        rows = report.load_rows(DATA)
        self.assertEqual(len(rows), 4)
        self.assertEqual(rows[0]["sku"], "A-100")
        self.assertEqual(rows[0]["location"], "Bay-1")


class LowStockTests(unittest.TestCase):
    def test_default_threshold(self):
        rows = report.load_rows(DATA)
        low = report.low_stock(rows)
        self.assertEqual([row["sku"] for row in low], ["A-100", "B-200", "B-201"])

    def test_custom_threshold(self):
        rows = report.load_rows(DATA)
        low = report.low_stock(rows, threshold=1)
        self.assertEqual([row["sku"] for row in low], ["B-200"])


class SummarizeTests(unittest.TestCase):
    def test_summary_counts(self):
        summary = report.summarize(report.load_rows(DATA))
        self.assertEqual(summary["total"], 4)
        self.assertEqual(summary["units"], 131)
        self.assertEqual(summary["low_stock"], 3)
        self.assertEqual(summary["unique_skus"], 4)


class GroupTests(unittest.TestCase):
    def test_group_by_location(self):
        groups = report.group_by_location(report.load_rows(DATA), {})
        self.assertEqual(sorted(groups), ["Bay-1", "Bay-2", "Bay-3"])
        self.assertEqual(len(groups["Bay-1"]), 2)


class FormatTests(unittest.TestCase):
    def test_report_mentions_totals(self):
        text = report.format_report(report.load_rows(DATA))
        self.assertIn("total lines: 4", text)
        self.assertIn("units on hand: 131", text)
        self.assertIn("low stock lines: 3", text)


if __name__ == "__main__":
    unittest.main()
```
