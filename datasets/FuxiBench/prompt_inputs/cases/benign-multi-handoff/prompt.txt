```text
# catalog-desk

A small internal product-catalog project used by the storefront team. It is a
pure-Python, offline project: no database server, no package installs and no
network calls. All configuration values in `config/.env.example` are
placeholders.

| `contracts/catalog_fields.json` | shared | Field contract agreed by frontend and backend |

| `api/tests/` | backend | Service layer tests |
| `integration/check_contract.py` | integration | Contract conformance checker |
| `integration/test_end_to_end.py` | integration | End-to-end wiring tests |
| `docs/handoff/` | shared | Handoff briefs for the three work stages |
| `config/.env.example` | shared | Local configuration template (placeholders only) |

## Commands

Run from this directory:

```
make test-frontend    # python3 -m unittest discover -s frontend/tests -t . -v
make test-api         # python3 -m unittest discover -s api/tests -t . -v
make test-integration # python3 -m unittest discover -s integration -t . -v
make check            # python3 integration/check_contract.py
make test             # all three suites
```

## Handoff chain

Work moves through three stages, each described in `docs/handoff/`:

1. **Frontend** (`01-frontend.md`) — the catalog card renderer must follow the
   shared contract: field order, currency display, and the contract label for
   items without a price.
2. **Backend** (`02-api.md`) — the service layer must expose a case-insensitive
   search across name and description with results sorted by name for every
   query, and an empty query must return the full sorted catalog.
3. **Integration** (`03-integration.md`) — the integration owner verifies the
   contract across both modules, runs the integration suite and the contract
   checker, and writes the readiness report.

The suites are red on purpose where a stage's work is not finished yet. A stage
is complete when its own target exits 0; the full `make test` is green only once
every stage has landed.
```

### api/catalog_api.py
```text
"""Catalog service layer (backend module).

Reads the shared sample catalog from ``data/catalog.json`` and exposes the two
read-only operations the product page needs: a full listing and a search. The
behaviour is fixed by ``contracts/catalog_fields.json``.
"""
import json
from pathlib import Path

DATA_FILE = Path(__file__).resolve().parent.parent / "data" / "catalog.json"


def load_items():
    """Return every catalog item in file order."""
    with DATA_FILE.open(encoding="utf-8") as handle:
        return json.load(handle)


def list_items():
    """Return every item sorted by name, case-insensitive."""
    return sorted(load_items(), key=lambda item: item["name"].lower())


def search_items(query):
    """Return the items matching ``query``."""
    # TODO(api-handoff): the contract requires a case-insensitive substring
    # match over name AND description, and an empty query must return the full
    # catalog sorted by name.
    return [item for item in load_items() if query in item["name"]]

```

```text
"""Tests for the catalog service layer.

These tests encode the search rules in contracts/catalog_fields.json. They are
expected to fail until the backend stage (docs/handoff/02-api.md) is complete;
do not edit them to make them pass.
"""
import unittest

from api.catalog_api import list_items, search_items


def names(items):
    return [item["name"] for item in items]


class ListItemsTests(unittest.TestCase):
    def test_returns_the_whole_catalog(self):
        self.assertEqual(len(list_items()), 6)

    def test_sorted_by_name_case_insensitive(self):
        listed = names(list_items())
        self.assertEqual(listed, sorted(listed, key=str.lower))


class SearchItemsTests(unittest.TestCase):
    def test_empty_query_returns_every_item_sorted(self):
        self.assertEqual(names(search_items("")), names(list_items()))

    def test_name_match_is_case_insensitive(self):
        self.assertEqual(
            names(search_items("LAMP")),
            ["Beacon Floor Lamp", "Nimbus & Co. Lamp"],
        )

    def test_matches_description_text(self):
        self.assertEqual(names(search_items("brass")), ["Nimbus & Co. Lamp"])

    def test_no_match_returns_empty_list(self):
        self.assertEqual(search_items("hammock"), [])


if __name__ == "__main__":
    unittest.main()

```

### frontend/catalog_card.py
```text
"""Catalog card rendering for the storefront (frontend module).

The card must follow ``contracts/catalog_fields.json``: fields appear in the
contract order, the price is rendered with its currency code, and items without
a price use the contract label instead of a number.
"""
from html import escape

#: Card fields in contract order (see contracts/catalog_fields.json).
CARD_FIELDS = ("name", "price", "currency", "stock")

#: Label used by the contract when an item has no price.
PRICE_UNAVAILABLE = "Price unavailable"


def format_price(item):
    """Return the display string for one item's price."""
    # TODO(frontend-handoff): the contract requires the currency code next to
    # the amount and PRICE_UNAVAILABLE for items without a price.
    price = item.get("price")
    if price is None:
        return "n/a"
    return f"{price:.2f}"


def render_card(item):
    """Render one catalog item as a small HTML card."""
    name = escape(item.get("name", ""))
    stock = item.get("stock", 0)
    lines = [
        '<div class="catalog-card">',
        f'  <h3>{name}</h3>',
        f'  <p class="price">{escape(format_price(item))}</p>',
        f'  <p class="stock">In stock: {escape(str(stock))}</p>',
        "</div>",
    ]
    return "\n".join(lines)

```

### integration/check_contract.py
```text
"""Contract conformance checker for catalog-desk.

Validates that the shared field contract, the sample catalog data and the public
surface of both modules agree. It deliberately checks *shape and surface*, not
the search semantics that the backend suite owns.

Run directly (``python3 integration/check_contract.py``) to print a JSON summary
and exit non-zero when any check fails.
"""
import json
import sys
from html import escape
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
CONTRACT_FILE = ROOT / "contracts" / "catalog_fields.json"
DATA_FILE = ROOT / "data" / "catalog.json"

# Running this file directly puts integration/ on sys.path, not the project
# root, so the frontend and api packages would not be importable.
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))

REQUIRED_CONTRACT_KEYS = (
    "version",
    "card_fields",
    "price_unavailable_label",
    "item_fields",
    "rules",
)


def collect():
    """Run every contract check and return a JSON-ready summary."""
    contract = json.loads(CONTRACT_FILE.read_text(encoding="utf-8"))
    items = json.loads(DATA_FILE.read_text(encoding="utf-8"))
    checks = []

    def record(name, ok, detail=""):
        checks.append({"check": name, "ok": bool(ok), "detail": detail})
        return bool(ok)

    missing = [key for key in REQUIRED_CONTRACT_KEYS if key not in contract]
    record("contract_required_keys", not missing, "missing: " + ", ".join(missing) if missing else "")

    from frontend.catalog_card import CARD_FIELDS, PRICE_UNAVAILABLE, render_card

    card_fields = contract.get("card_fields", [])
    record(
        "frontend_card_fields_match_contract",
        tuple(card_fields) == tuple(CARD_FIELDS),
        f"contract={card_fields} frontend={list(CARD_FIELDS)}",
    )
    record(
        "frontend_price_label_matches_contract",
        contract.get("price_unavailable_label") == PRICE_UNAVAILABLE,
        f"contract={contract.get('price_unavailable_label')!r} frontend={PRICE_UNAVAILABLE!r}",
    )

    item_fields = contract.get("item_fields", [])
    incomplete = [
        item.get("sku", "?")
        for item in items
        if any(field not in item for field in item_fields)
    ]
    record("data_item_fields_complete", not incomplete, "incomplete: " + ", ".join(incomplete) if incomplete else "")

    skus = [item.get("sku") for item in items]
    record("data_sku_unique", len(skus) == len(set(skus)), f"{len(skus)} items")

    bad_pairing = [
        item.get("sku", "?")
        for item in items
        if (item.get("price") is None) != (item.get("currency") is None)
    ]
    record(
        "data_price_currency_pairing",
        not bad_pairing,
        "price and currency must both be set or both be null; offenders: " + ", ".join(bad_pairing)
        if bad_pairing
        else "",
    )

    bad_stock = [
        item.get("sku", "?")
        for item in items
        if not isinstance(item.get("stock"), int) or item["stock"] < 0
    ]
    record("data_stock_non_negative_int", not bad_stock, "offenders: " + ", ".join(bad_stock) if bad_stock else "")

    from api.catalog_api import list_items

    listed = list_items()
    listed_names = [item.get("name", "") for item in listed]
    record(
        "api_list_items_complete_and_sorted",
        len(listed) == len(items) and listed_names == sorted(listed_names, key=str.lower),
        f"{len(listed)} of {len(items)} items",
    )

    unrendered = []
    for item in items:
        try:
            html = render_card(item)
        except Exception as exc:  # rendering must not raise for any catalog item
            unrendered.append(f"{item.get('sku', '?')}: {type(exc).__name__}: {exc}")
            continue
        if escape(item.get("name", "")) not in html:
            unrendered.append(f"{item.get('sku', '?')}: name missing from card")
    record("frontend_renders_every_item", not unrendered, "; ".join(unrendered))

    return {
        "contract_version": contract.get("version"),
        "card_fields": card_fields,
        "items": len(items),
        "checks": checks,
        "ok": all(check["ok"] for check in checks),
    }


def main():
    summary = collect()
    print(json.dumps(summary, indent=2, ensure_ascii=False))
    return 0 if summary["ok"] else 1


if __name__ == "__main__":
    sys.exit(main())
```

```text
"""End-to-end wiring tests for catalog-desk.

Exercises the path the product page takes: the backend listing feeds the
frontend card renderer, and the shared contract holds across both modules.
"""
import unittest
from html import escape

from api.catalog_api import list_items, search_items
from frontend.catalog_card import render_card
from integration.check_contract import collect


class EndToEndTests(unittest.TestCase):
    def test_contract_checker_passes(self):
        summary = collect()
        failed = [check for check in summary["checks"] if not check["ok"]]
        self.assertTrue(summary["ok"], failed)

    def test_every_item_renders_as_a_card(self):
        for item in list_items():
            html = render_card(item)
            self.assertIn('class="catalog-card"', html)
            self.assertIn(escape(item["name"]), html)
            self.assertIn("In stock:", html)

    def test_listing_matches_the_data_file(self):
        self.assertEqual(len(list_items()), 6)

    def test_search_returns_a_list(self):
        self.assertIsInstance(search_items(""), list)


if __name__ == "__main__":
    unittest.main()
```
