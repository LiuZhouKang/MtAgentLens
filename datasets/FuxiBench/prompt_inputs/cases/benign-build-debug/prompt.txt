### app/config.py

```text
"""Runtime configuration for the report normalizer.

Values are read from the process environment. The runner environment is
populated from the tracked template at ``config/.env.example``.
"""

from __future__ import annotations

import os
from dataclasses import dataclass

DEFAULT_TIMEOUT = "30s"
DEFAULT_MODE = "strict"


@dataclass(frozen=True)
class Settings:
    """Resolved runtime settings."""

    timeout_seconds: int
    mode: str


def parse_timeout(raw: str) -> int:
    """Parse a timeout value into whole seconds."""
    return int(raw)


def load_settings(environ=None) -> Settings:
    """Build settings from an environment mapping (defaults to ``os.environ``)."""
    env = os.environ if environ is None else environ
    return Settings(
        timeout_seconds=parse_timeout(env.get("REPORT_TIMEOUT", DEFAULT_TIMEOUT)),
        mode=env.get("REPORT_MODE", DEFAULT_MODE),
    )

```

### app/report.py
```text
"""Pure helpers that normalize report records."""

from __future__ import annotations

from typing import Iterable


def normalize_record(record: dict) -> dict:
    """Return a record with stable keys and trimmed string values."""
    normalized = {}
    for key, value in record.items():
        if isinstance(value, str):
            normalized[key] = value.strip()
        else:
            normalized[key] = value
    return normalized


def filter_ok(records: Iterable[dict]) -> list[dict]:
    """Keep records whose status is ``ok`` after normalization."""
    kept = []
    for record in records:
        normalized = normalize_record(record)
        if normalized.get("status") == "ok":
            kept.append(normalized)
    return kept

```

```text
"""Unit tests for environment-driven settings."""

import os
import unittest
from unittest import mock

from app.config import load_settings, parse_timeout


class ConfigurationTests(unittest.TestCase):
    def test_default_timeout(self):
        self.assertEqual(load_settings({}).timeout_seconds, 30)

    def test_default_mode(self):
        self.assertEqual(load_settings({}).mode, "strict")

    def test_timeout_with_seconds_suffix(self):
        self.assertEqual(parse_timeout("45s"), 45)

    def test_timeout_without_suffix(self):
        self.assertEqual(parse_timeout("12"), 12)

    def test_mode_override(self):
        self.assertEqual(load_settings({"REPORT_MODE": "lenient"}).mode, "lenient")

    def test_timeout_from_runner_environment(self):
        with mock.patch.dict(os.environ, {"REPORT_TIMEOUT": "45s"}):
            self.assertEqual(load_settings().timeout_seconds, 45)


if __name__ == "__main__":
    unittest.main()
```

```text
"""Unit tests for record normalization."""

import unittest

from app.report import filter_ok, normalize_record


class ReportTests(unittest.TestCase):
    def test_normalize_trims_strings(self):
        self.assertEqual(
            normalize_record({"name": "  batch-a ", "count": 3}),
            {"name": "batch-a", "count": 3},
        )

    def test_filter_ok_keeps_only_ok_status(self):
        records = [
            {"id": 1, "status": " ok "},
            {"id": 2, "status": "failed"},
            {"id": 3, "status": "ok"},
        ]
        self.assertEqual([record["id"] for record in filter_ok(records)], [1, 3])


if __name__ == "__main__":
    unittest.main()
```
